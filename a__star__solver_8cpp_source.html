76std::optional<RouteResult> findRoute(
77 const domain::Graph& graph,
78 domain::NodeId start,
79 domain::NodeId goal,
80 const IRouteStrategy& strategy,
81 const domain::TransportParams& params,
82 std::chrono::system_clock::time_point departure
83) {
84 if (!graph.hasNode(start) || !graph.hasNode(goal)) {
85 return std::nullopt;
86 }
87
88 auto cmp = [&strategy](const OpenNode& a, const OpenNode& b) {
89 return strategy.less(b.f, a.f);
90 };
91 std::priority_queue<OpenNode, std::vector<OpenNode>, decltype(cmp)> open(cmp);
92
93 std::unordered_map<StateKey, domain::CombinedCost, StateKeyHash> best_g;
94 std::unordered_map<StateKey, TransitionInfo, StateKeyHash> came_from;
95
96 domain::CombinedCost g0;
97 domain::CombinedCost h0 = strategy.heuristic(graph.getNode(start), graph.getNode(goal), params);
98 StateKey start_key{start, false, false};
99 open.push(OpenNode{start, g0, g0 + h0, departure, std::nullopt, false, false});
100 best_g[start_key] = g0;
101
102 while (!open.empty()) {
103 OpenNode cur = open.top();
104 open.pop();
105
106 StateKey cur_key{cur.id, cur.has_public_ticket, cur.has_left_car};
107
108 if (cur.id == goal) {
109 // Восстановление пути
110 std::vector<domain::NodeId> path_nodes;
111 std::vector<domain::EdgeId> path_edges;
112 std::vector<domain::TransportType> path_transports;
113 StateKey k = cur_key;
114 path_nodes.push_back(k.node);
115 while (came_from.count(k) > 0) {
116 auto& info = came_from[k];
117 path_edges.push_back(info.edge_id);
118 path_transports.push_back(info.actual_transport);
119 path_nodes.push_back(info.prev_key.node);
120 k = info.prev_key;
121 }
122 std::reverse(path_nodes.begin(), path_nodes.end());
123 std::reverse(path_edges.begin(), path_edges.end());
124 std::reverse(path_transports.begin(), path_transports.end());
125 return RouteResult{path_nodes, path_edges, path_transports, cur.g, {}};
126 }
127
128 auto bg_it = best_g.find(cur_key);
129 if (bg_it != best_g.end() && strategy.less(bg_it->second, cur.g)) {
130 continue;
131 }
132
133 for (const domain::Edge& edge : graph.getEdgesFrom(cur.id)) {
134 // Определяем фактический тип транспорта
135 domain::TransportType actual = actualTransportType(edge.transport, cur.has_public_ticket);
136
137 // Личный автомобиль запрещён, если куплен билет на общ. транспорт.
138 // Но по Car-ребрам можно проехать на такси (actual == Taxi).
139 // Если actual == Car - значит билета нет, едем на личном авто.
140 // Если actual == Taxi - значит билет есть, едем на такси по Car-ребру.
141
142 // Нельзя вернуться на личный авто после выхода из него.
143 if (actual == domain::TransportType::Car && cur.has_left_car) {
144 continue;
145 }
146
147 domain::QueryContext ctx{cur.arrival, params};
148 auto res = edge.logic->calculate(edge, ctx, strategy);
149 if (!res) continue;
150
151 // Пересчитываем стоимость, если фактический транспорт - такси
152 domain::CombinedCost edge_cost = res->cost;
153 if (actual == domain::TransportType::Taxi && edge.transport == domain::TransportType::Car) {
154 // Заменяем стоимость Car на стоимость Taxi
155 double distance_km = edge.length_meters / 1000.0;
156 edge_cost.money_rub = params.taxi_base_price + distance_km * params.taxi_price_per_km;
157 }
158
159 // Подсчёт пересадок:
160 // - Car/Taxi: смена дороги не пересадка
161 // - Bus/Metro: смена маршрута = пересадка
162 // - Смена вида транспорта (Taxi->Bus, Bus->Metro и т.п.) = пересадка
163 int transfer_add = 0;
164 if (isPublicTransport(actual)) {
165 // Общ. транспорт: пересадка при смене маршрута
166 if (cur.last_route) {
167 auto& prev = *cur.last_route;
168 if (prev.transport != actual || prev.name != edge.name) {
169 transfer_add = 1;
170 }
171 }
172 } else if (actual == domain::TransportType::Taxi) {
173 // Такси: пересадка только при смене на общ. транспорт
174 // (Taxi->Taxi не пересадка)
175 if (cur.last_route && isPublicTransport(cur.last_route->transport)) {
176 transfer_add = 1;
177 }
178 }
179 // Car и Walk: пересадок нет
180
181 edge_cost.transfers += transfer_add;
182
183 // Безлимитный билет на общественный транспорт:
184 // покупается один раз при первом использовании Bus/Metro,
185 // действует на все виды общ. транспорта на всём маршруте.
186 bool next_has_ticket = cur.has_public_ticket;
187 if (isPublicTransport(edge.transport)) {
188 if (cur.has_public_ticket) {
189 // Билет уже куплен - общ. транспорт бесплатный
190 edge_cost.money_rub = 0.0;
191 } else {
192 // Первое использование общ. транспорта - покупаем билет
193 next_has_ticket = true;
194 }
195 }
196
197 domain::CombinedCost new_g = cur.g + edge_cost;
198
199 // Отслеживание выхода из личного авто:
200 // has_left_car = true, если мы когда-то ехали на Car и теперь не на Car.
201 // Taxi не считается - на такси можно переключаться свободно.
202 bool next_has_left_car = cur.has_left_car;
203 if (actual == domain::TransportType::Car) {
204 // Сейчас едем на личном авто - ещё не вышли
205 next_has_left_car = false;
206 } else if (cur.last_route && cur.last_route->transport == domain::TransportType::Car) {
207 // Были на личном авто и теперь переключились - вышли из авто
208 next_has_left_car = true;
209 }
210 // Если уже has_left_car, оно сохраняется
211
212 StateKey next_key{edge.to, next_has_ticket, next_has_left_car};
213 auto it = best_g.find(next_key);
214 if (it == best_g.end() || strategy.less(new_g, it->second)) {
215 best_g[next_key] = new_g;
216 came_from[next_key] = TransitionInfo{cur_key, edge.id, actual};
217
218 std::optional<LastRoute> next_last = cur.last_route;
219 if (actual != domain::TransportType::Walk) {
220 next_last = LastRoute{actual, edge.name};
221 }
222
223 domain::CombinedCost h = strategy.heuristic(graph.getNode(edge.to), graph.getNode(goal), params);
224 open.push(OpenNode{
225 edge.to,
226 new_g,
227 new_g + h,
228 res->arrival_time,
229 next_last,
230 next_has_ticket,
231 next_has_left_car
232 });
233 }
234 }
235 }
236
237 return std::nullopt;
238}
