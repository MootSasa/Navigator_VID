182std::unique_ptr<domain::Graph> GraphBuilder::build() {
183 // Валидация данных перед сборкой
184 validateData();
185
186 // Создаём пустой граф
187 auto graph = std::make_unique<domain::Graph>();
188
189 // Добавляем все узлы в граф
190 for (const auto& parsed_node : nodes_) {
191 domain::Node node;
192 node.id = parsed_node.id;
193 node.coords.x = parsed_node.x;
194 node.coords.y = parsed_node.y;
195
196 // Определяем тип узла
197 if (parsed_node.type == "intersection") {
198 node.type = domain::NodeType::Intersection;
199 } else if (parsed_node.type == "bus_stop") {
200 node.type = domain::NodeType::BusStop;
201 } else if (parsed_node.type == "metro_station") {
202 node.type = domain::NodeType::MetroStation;
203 } else if (parsed_node.type == "poi") {
204 node.type = domain::NodeType::Poi;
205 } else {
206 // По умолчанию - перекрёсток
207 node.type = domain::NodeType::Intersection;
208 }
209
210 node.name = parsed_node.name;
211 graph->addNode(node);
212 }
213
214 // Добавляем все рёбра в граф
215 for (const auto& parsed_edge : edges_) {
216 domain::Edge edge;
217 edge.id = parsed_edge.id;
218 edge.from = parsed_edge.from_node;
219 edge.to = parsed_edge.to_node;
220 edge.name = parsed_edge.name;
221 edge.length_meters = parsed_edge.length_meters;
222
223 // Определяем тип транспорта
224 if (parsed_edge.transport_type == "walk") {
225 edge.transport = domain::TransportType::Walk;
226 } else if (parsed_edge.transport_type == "car") {
227 edge.transport = domain::TransportType::Car;
228 } else if (parsed_edge.transport_type == "bus") {
229 edge.transport = domain::TransportType::Bus;
230 } else if (parsed_edge.transport_type == "metro") {
231 edge.transport = domain::TransportType::Metro;
232 } else if (parsed_edge.transport_type == "taxi") {
233 edge.transport = domain::TransportType::Taxi;
234 } else {
235 // По умолчанию - пешком
236 edge.transport = domain::TransportType::Walk;
237 }
238
239 // Создаём логику для ребра на основе расписания.
240 // Пока используем StaticLogic для всех рёбер.
241 // ScheduledLogic, FrequencyBasedLogic и TimeWindowLogic
242 // будут добавлены на этапе 2 разработки.
243
244 // Ищем, есть ли для этого ребра особое расписание
245 auto sched_it = schedules_.find(edge.id);
246
247 if (edge.transport == domain::TransportType::Metro) {
248 // Метро не имеет расписания в schedules.csv, у него частотная логика
249 edge.logic = std::make_unique<domain::FrequencyBasedLogic>();
250
251 } else if (sched_it != schedules_.end()) {
252 if (sched_it->second.schedule_type == "time_window") {
253 // Логика разводных мостов
254 try {
255 int close_h = std::stoi(sched_it->second.params.substr(0, 2));
256 int open_h = std::stoi(sched_it->second.params.substr(6, 2));
257 edge.logic = std::make_unique<domain::TimeWindowLogic>(close_h, open_h);
258 } catch (...) {
259 edge.logic = std::make_unique<domain::StaticLogic>();
260 }
261
262 } else if (sched_it->second.schedule_type == "fixed_times") {
263 // Логика автобусов: парсим "08:00,08:15,09:00" в минуты
264 std::vector<int> departure_minutes;
265 std::stringstream ss(sched_it->second.params);
266 std::string time_str;
267
268 while (std::getline(ss, time_str, ',')) {
269 try {
270 int h = std::stoi(time_str.substr(0, 2));
271 int m = std::stoi(time_str.substr(3, 2));
272 departure_minutes.push_back(h * 60 + m);
273 } catch (...) {}
274 }
275 // На всякий случай сортируем расписание
276 std::sort(departure_minutes.begin(), departure_minutes.end());
277 edge.logic = std::make_unique<domain::ScheduledLogic>(departure_minutes);
278
279 } else {
280 edge.logic = std::make_unique<domain::StaticLogic>();
281 }
282 } else {
283 // Для всех остальных (пешком, авто без мостов) - статичная логика
284 edge.logic = std::make_unique<domain::StaticLogic>();
285 }
286
287 graph->addEdge(std::move(edge));
288 }
289
290 return graph;
291}
