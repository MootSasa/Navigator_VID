101 if (result.nodes.empty()) return;
102
103 // Строим сегменты по узлам маршрута
104 for (size_t i = 0; i + 1 < result.nodes.size(); ++i) {
105 const auto& from = graph.getNode(result.nodes[i]);
106 const auto& to = graph.getNode(result.nodes[i + 1]);
107 QPointF p1(from.coords.x, -from.coords.y);
108 QPointF p2(to.coords.x, -to.coords.y);
109 segments_.append(QLineF(p1, p2));

111
112 // Позиции маркеров

115 startPos_ = QPointF(startNode.coords.x, -startNode.coords.y);

118 update();
